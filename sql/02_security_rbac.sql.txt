
-- =========================================
-- 1. Create Database Roles
-- =========================================

DO $$
BEGIN
    IF NOT EXISTS (
        SELECT 1 FROM pg_roles WHERE rolname = 'sacco_teller'
    ) THEN
        CREATE ROLE sacco_teller;
    END IF;

    IF NOT EXISTS (
        SELECT 1 FROM pg_roles WHERE rolname = 'sacco_manager'
    ) THEN
        CREATE ROLE sacco_manager;
    END IF;
END
$$;


-- =========================================
-- 2. Grant Minimal Permissions to Tellers
-- =========================================

GRANT SELECT, INSERT ON TABLE Members
TO sacco_teller;

GRANT SELECT, UPDATE ON TABLE SavingsAccounts
TO sacco_teller;

GRANT SELECT, INSERT ON TABLE Transactions
TO sacco_teller;


-- =========================================
-- 3. Grant Sequence Permissions to Tellers
-- =========================================

GRANT USAGE, SELECT
ON ALL SEQUENCES IN SCHEMA public
TO sacco_teller;


-- =========================================
-- 4. Grant Administrative Permissions to Managers
-- =========================================

GRANT ALL PRIVILEGES
ON ALL TABLES IN SCHEMA public
TO sacco_manager;

GRANT ALL PRIVILEGES
ON ALL SEQUENCES IN SCHEMA public
TO sacco_manager;


-- =========================================
-- 5. Set Default Permissions for Future Tables
-- =========================================

ALTER DEFAULT PRIVILEGES IN SCHEMA public
GRANT ALL PRIVILEGES ON TABLES TO sacco_manager;

ALTER DEFAULT PRIVILEGES IN SCHEMA public
GRANT ALL PRIVILEGES ON SEQUENCES TO sacco_manager;
