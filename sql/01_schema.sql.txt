
-- ==========================================
-- 8. Additional Data Integrity Constraints
-- ==========================================

-- Validate loan interest rates
ALTER TABLE Loans
ADD CONSTRAINT chk_loan_interest_rate
CHECK (InterestRate BETWEEN 0 AND 100);

-- Validate repayment amounts
ALTER TABLE Repayments
ADD CONSTRAINT chk_amount_due_positive
CHECK (AmountDue > 0);

ALTER TABLE Repayments
ADD CONSTRAINT chk_amount_paid_valid
CHECK (AmountPaid >= 0 AND AmountPaid <= AmountDue);

-- Ensure payment date matches payment amount
ALTER TABLE Repayments
ADD CONSTRAINT chk_payment_date
CHECK (
    (AmountPaid = 0 AND PaymentDate IS NULL)
    OR
    (AmountPaid > 0 AND PaymentDate IS NOT NULL)
);

-- A paid repayment must be fully paid
ALTER TABLE Repayments
ADD CONSTRAINT chk_paid_repayment
CHECK (
    Status <> 'Paid'
    OR AmountPaid = AmountDue
);
